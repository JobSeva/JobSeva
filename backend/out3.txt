import { prisma } from "../lib/prisma";
import { AppError } from "../common/errors";
import { Application, ApplicationStatus } from "../types/domain";

interface ListApplicationsQuery {
  status?: ApplicationStatus;
  page: number;
  limit: number;
}

const createMatchScore = (jobId: string): number => {
  let seed = 0;
  for (const char of jobId) {
    seed += char.charCodeAt(0);
  }
  return 65 + (seed % 31);
};

export const applicationsService = {
  async listForSeeker(
    seekerId: string,
    query: ListApplicationsQuery,
  ): Promise<{ items: Application[]; total: number; page: number; limit: number }> {
    const where: any = { seekerId };
    if (query.status) {
      where.status = query.status;
    }

    const total = await prisma.application.count({ where });

    const dbApplications = await prisma.application.findMany({
      where,
      orderBy: { updatedAt: "desc" },
      skip: (query.page - 1) * query.limit,
      take: query.limit,
    });

    const items = dbApplications.map((app) => mapApplicationRecord(app));

    return {
      items,
      total,
      page: query.page,
      limit: query.limit,
    };
  },

  async create(seekerId: string, jobId: string): Promise<Application> {
    const job = await prisma.job.findUnique({
      where: { id: jobId },
    });

    if (!job || !job.active) {
      throw new AppError(404, "Job not found", "JOB_NOT_FOUND");
    }

    const existing = await prisma.application.findFirst({
      where: {
        seekerId,
        jobId,
      },
    });

    if (existing) {
      throw new AppError(
        409,
        "You have already applied to this job",
        "APPLICATION_EXISTS",
      );
    }

    // Get company profile to retrieve name and logo
    const company = await prisma.companyProfile.findUnique({
      where: { companyId: job.companyId },
    });

    const application = await prisma.application.create({
      data: {
        seekerId,
        jobId,
        jobTitle: job.title,
        company: company?.name || "",
        companyLogo: company?.logo || "",
        status: "applied",
        matchScore: createMatchScore(jobId),
      },
    });

    // Increment job applicants count
    await prisma.job.update({
      where: { id: jobId },
      data: { applicantsCount: { increment: 1 } },
    });

    return mapApplicationRecord(application);
  },

  async getByIdForSeeker(
    seekerId: string,
    applicationId: string,
  ): Promise<Application> {
    const application = await prisma.application.findFirst({
      where: {
        id: applicationId,
        seekerId,
      },
    });

    if (!application) {
      throw new AppError(404, "Application not found", "APPLICATION_NOT_FOUND");
    }

    return mapApplicationRecord(application);
  },

  async withdraw(seekerId: string, applicationId: string): Promise<void> {
    const application = await prisma.application.findFirst({
      where: {
        id: applicationId,
        seekerId,
      },
    });

    if (!application) {
      throw new AppError(404, "Application not found", "APPLICATION_NOT_FOUND");
    }

    if (application.status === "hired") {
      throw new AppError(
        409,
        "Cannot withdraw an application that is already hired",
        "INVALID_APPLICATION_STATE",
      );
    }

    // Delete the application
    await prisma.application.delete({
      where: { id: applicationId },
    });

    // Decrement job applicants count
    if (application.jobId) {
      const job = await prisma.job.findUnique({
        where: { id: application.jobId },
      });
      if (job && job.applicantsCount > 0) {
        await prisma.job.update({
          where: { id: application.jobId },
          data: { applicantsCount: { decrement: 1 } },
        });
      }
    }
  },
};

function mapApplicationRecord(record: any): Application {
  return {
    id: record.id,
    seekerId: record.seekerId,
    jobId: record.jobId,
    jobTitle: record.jobTitle,
    company: record.company,
    companyLogo: record.companyLogo,
    status: record.status as ApplicationStatus,
    appliedAt: record.appliedAt.toISOString(),
    updatedAt: record.updatedAt.toISOString(),
    matchScore: record.matchScore,
    recruiterRating: record.recruiterRating || undefined,
    recruiterNote: record.recruiterNote || undefined,
  };
}

import { prisma } from "../lib/prisma";
import { Message, Notification } from "../types/domain";

export class CommunicationsService {
  // === Messages === //

  async getMessagesForUser(userId: string): Promise<Message[]> {
    const dbMessages = await prisma.message.findMany({
      where: {
        OR: [
          { senderId: userId },
          { receiverId: userId },
        ],
      },
      orderBy: { createdAt: "asc" },
    });

    return dbMessages.map((m) => mapMessageRecord(m));
  }

  async getConversations(
    userId: string,
  ): Promise<Array<{ otherUserId: string; latestMessage: Message }>> {
    const messages = await prisma.message.findMany({
      where: {
        OR: [
          { senderId: userId },
          { receiverId: userId },
        ],
      },
      orderBy: { createdAt: "desc" },
    });

    // Group by the other user and keep most recent message
    const conversationMap = new Map<string, any>();

    for (const msg of messages) {
      const otherUserId = msg.senderId === userId ? msg.receiverId : msg.senderId;

      if (!conversationMap.has(otherUserId)) {
        conversationMap.set(otherUserId, msg);
      }
    }

    const result = Array.from(conversationMap.entries()).map(
      ([otherUserId, msg]) => ({
        otherUserId,
        latestMessage: mapMessageRecord(msg),
      }),
    );

    return result;
  }

  async getMessages(
    userId: string,
    otherUserId: string,
  ): Promise<Message[]> {
    const dbMessages = await prisma.message.findMany({
      where: {
        OR: [
          {
            senderId: userId,
            receiverId: otherUserId,
          },
          {
            senderId: otherUserId,
            receiverId: userId,
          },
        ],
      },
      orderBy: { createdAt: "asc" },
    });

    return dbMessages.map((m) => mapMessageRecord(m));
  }

  async getUnreadMessagesCount(userId: string): Promise<number> {
    return prisma.message.count({
      where: {
        receiverId: userId,
        isRead: false,
      },
    });
  }

  async sendMessage(
    senderId: string,
    receiverId: string,
    content: string,
    applicationId?: string,
  ): Promise<Message> {
    const message = await prisma.message.create({
      data: {
        senderId,
        receiverId,
        content,
        applicationId: applicationId || null,
        isRead: false,
      },
    });

    return mapMessageRecord(message);
  }

  async markMessageAsRead(messageId: string, userId: string): Promise<Message> {
    const message = await prisma.message.findUnique({
      where: { id: messageId },
    });

    if (!message) throw new Error("Message not found");
    if (message.receiverId !== userId) throw new Error("Not authorized");

    const updated = await prisma.message.update({
      where: { id: messageId },
      data: { isRead: true },
    });

    return mapMessageRecord(updated);
  }

  // === Notifications === //

  async getNotificationsForUser(userId: string): Promise<Notification[]> {
    const dbNotifications = await prisma.notification.findMany({
      where: { userId },
      orderBy: { createdAt: "desc" },
    });

    return dbNotifications.map((n) => mapNotificationRecord(n));
  }

  async getUnreadNotificationsCount(userId: string): Promise<number> {
    return prisma.notification.count({
      where: {
        userId,
        isRead: false,
      },
    });
  }

  async createNotification(
    userId: string,
    title: string,
    body: string,
    type: "application" | "message" | "system",
  ): Promise<Notification> {
    const notification = await prisma.notification.create({
      data: {
        userId,
        title,
        body,
        type,
        isRead: false,
      },
    });

    return mapNotificationRecord(notification);
  }

  async markNotificationAsRead(
    notificationId: string,
    userId: string,
  ): Promise<Notification> {
    const notification = await prisma.notification.findUnique({
      where: { id: notificationId },
    });

    if (!notification) throw new Error("Notification not found");
    if (notification.userId !== userId) throw new Error("Not authorized");

    const updated = await prisma.notification.update({
      where: { id: notificationId },
      data: { isRead: true },
    });

    return mapNotificationRecord(updated);
  }

  async markAllNotificationsAsRead(
    userId: string,
  ): Promise<{ success: boolean; count: number }> {
    const result = await prisma.notification.updateMany({
      where: {
        userId,
        isRead: false,
      },
      data: { isRead: true },
    });

    return { success: true, count: result.count };
  }
}

export const commsService = new CommunicationsService();

function mapMessageRecord(record: any): Message {
  return {
    id: record.id,
    senderId: record.senderId,
    receiverId: record.receiverId,
    applicationId: record.applicationId || undefined,
    content: record.content,
    isRead: record.isRead,
    createdAt: record.createdAt.toISOString(),
  };
}

function mapNotificationRecord(record: any): Notification {
  return {
    id: record.id,
    userId: record.userId,
    title: record.title,
    body: record.body,
    type: record.type as "application" | "message" | "system",
    isRead: record.isRead,
    createdAt: record.createdAt.toISOString(),
  };
}

```typescript
import { Response } from "express";

import { AppError } from "../common/errors";
import { success } from "../common/http";
import { applicationsService } from "../services/applications.service";
import { ApplicationStatus } from "../types/domain";
import { AuthenticatedRequest } from "../types/auth";

const requireSeekerAuth = (req: AuthenticatedRequest): string => {
  if (!req.auth || req.auth.role !== "seeker") {
    throw new AppError(403, "Seeker access required", "FORBIDDEN");
  }

  return req.auth.userId;
};

export const applicationsController = {
  async list(req: AuthenticatedRequest, res: Response): Promise<void> {
    const seekerId = requireSeekerAuth(req);
    const query = req.query as Record<string, string | undefined>;

    const status =
      query.status === "applied" ||
      query.status === "shortlisted" ||
      query.status === "interview" ||
      query.status === "hired" ||
      query.status === "rejected"
        ? (query.status as ApplicationStatus)
        : undefined;

    const page = query.page ? Math.max(1, Number(query.page) || 1) : 1;
    const limit = query.limit
      ? Math.min(100, Math.max(1, Number(query.limit) || 10))
      : 10;

    const result = await applicationsService.listForSeeker(seekerId, {
      status,
      page,
      limit,
    });

    res.status(200).json(
      success(result.items, {
        total: result.total,
        page: result.page,
        limit: result.limit,
        totalPages: Math.ceil(result.total / result.limit),
      }),
    );
  },

  async create(req: AuthenticatedRequest, res: Response): Promise<void> {
    const seekerId = requireSeekerAuth(req);
    const { jobId } = req.body as { jobId: string };

    const application = await applicationsService.create(seekerId, jobId);
    res.status(201).json(success(application));
  },

  async getById(req: AuthenticatedRequest, res: Response): Promise<void> {
    const seekerId = requireSeekerAuth(req);
    const { applicationId } = req.params;

    if (!applicationId) {
      throw new AppError(400, "applicationId is required", "VALIDATION_ERROR");
    }

    const application = await applicationsService.getByIdForSeeker(
      seekerId,
      applicationId,
    );
    res.status(200).json(success(application));
  },

  async withdraw(req: AuthenticatedRequest, res: Response): Promise<void> {
    const seekerId = requireSeekerAuth(req);
    const { applicationId } = req.params;

    if (!applicationId) {
      throw new AppError(400, "applicationId is required", "VALIDATION_ERROR");
    }

    await applicationsService.withdraw(seekerId, applicationId);
    res.status(200).json(success({ message: "Application withdrawn" }));
  },
};
```
